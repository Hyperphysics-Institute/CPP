\section{The discrete-symmetry character of the four factors}
\label{sec:parities}
% ============================================================

We track two discrete symmetries of the substrate: spatial parity $\Po$ (the enantiomorph swap
$\nhat \to -\nhat$) and time reversal $\Te$ (reversal of the Absolute-Moment ordering, hence of the
PCD progression and of DI-bit propagation direction). A quantity is $\Po$-even/odd ($\Te$-even/odd)
according to its sign under $\Po$ ($\Te$).

\begin{lemma}[$\nhat$ is $\Po$-odd, $\Te$-even; $\sgn(\nhat) = \text{FI-C-9}$]
\label{lem:nhat}
The substrate primitive direction $\nhat = \vhost/|\vhost|$ is a polar spatial vector: under $\Po$,
$\nhat \to -\nhat$ ($\Po$-odd); it is unchanged under $\Te$ ($\Te$-even). By the framework
identification FI-C-RC-1 / FI-C-9, the choice of $\nhat$ versus $-\nhat$ is the frozen
substrate-vacuum enantiomorph: $\sgn(\nhat) = \text{FI-C-9}$, a pseudoscalar ($\Po$-odd, $\Te$-even),
consistent with $\Po$ flipping the handedness.
\end{lemma}

\begin{lemma}[$\delta$ is $\Po$-even]
\label{lem:delta-p}
Mechanism A (MA.1) gives the propagation rate $r(\hat{e}) = r_0(1 + \delta\,\hat{e}\cdot\nhat)$. A
propagation rate is a $\Po$-even scalar (a count of events per Absolute Moment, unchanged by a
spatial reflection). Under $\Po$ both $\hat{e}\to-\hat{e}$ and $\nhat\to-\nhat$, so the contraction
$\hat{e}\cdot\nhat$ is $\Po$-even. For $r$ to be $\Po$-even, $\delta$ must be $\Po$-even. In
particular $\sgn(\delta)$ does \emph{not} flip under the enantiomorph swap.
\end{lemma}

\begin{lemma}[$\delta$ is $\Te$-odd]
\label{lem:delta-t}
Under time reversal, the elementary process ``a DI-bit propagates along $\hat{e}$ at rate
$r(\hat{e})$'' maps to ``a DI-bit propagates along $-\hat{e}$,'' so $\Te$-covariance of the
dynamics requires the time-reversed rate function $r^{\Te}(\hat{e}) = r(-\hat{e}) =
r_0(1 - \delta\,\hat{e}\cdot\nhat) = r_0(1 + (-\delta)\,\hat{e}\cdot\nhat)$. Hence the $\Te$-image
parameter is $-\delta$: $\delta$ is $\Te$-odd. (Physically: a directional bias in propagation is an
arrow-like quantity, like a drift or a current bias.) Consistency check: $\jnet =
(6\delta/\varphi^2)\nhat$ is a current, $\Te$-odd ($\jnet \to -\jnet$); with $\delta$ $\Te$-odd and
$\nhat$ $\Te$-even this is satisfied.
\end{lemma}

\begin{lemma}[$\wPCD$ is $\Po$-even (axial), $\Te$-odd]
\label{lem:omega}
By THEO-CHIR-PCD-ORIENTATION-1 (and A6$'$), $\wPCD$ is an \emph{axial} vector (pseudovector): the
axial vector associated with the Perceive$\to$Compute$\to$Displace progression sense. Axial vectors
are $\Po$-even. The progression sense reverses under time reversal (the cycle runs backwards), so
$\wPCD$ is $\Te$-odd. (A circulation/orientation sense is the standard example of a $\Po$-even,
$\Te$-odd axial vector, like angular momentum's orientation under $\Te$.)
\end{lemma}

\begin{lemma}[$\scyc$ is $\Po$-odd and $\Te$-odd]
\label{lem:sigma}
From $\wPCD = \scyc\,\nhat$ with $\wPCD$ $\Po$-even (Lemma~\ref{lem:omega}) and $\nhat$ $\Po$-odd
(Lemma~\ref{lem:nhat}), $\scyc$ is $\Po$-odd: it is a pseudoscalar. From $\wPCD$ $\Te$-odd and
$\nhat$ $\Te$-even, $\scyc$ is $\Te$-odd. (The $\Te$-odd character matches MERGE-1's statement that
the two $\sigma$-choices are time-reversal-symmetric framings.)
\end{lemma}

\begin{lemma}[The 600-cell is achiral: FI-C-9 is the unique primitive pseudoscalar]
\label{lem:achiral}

\section{The parity decomposition theorem}
\label{sec:theorem}
% ============================================================

\begin{theorem}[Parity decomposition of the cycle handedness; THEO-CHIR-MERGE-2]
\label{thm:main}
Under CPP axioms A1--A11 $+$ Reading C $+$ Mechanism A (sketch-document Layer 3), with the
discrete-symmetry characters of \S\ref{sec:parities}, the temporal cycle handedness
\eqref{eq:merge1-relation} decomposes by symmetry character as
\[
\underbrace{\scyc}_{\Po\text{-odd},\ \Te\text{-odd}}
\;=\;
\underbrace{\sgn(\delta)}_{\Po\text{-even},\ \Te\text{-odd}}
\;\cdot\;
\underbrace{\sgn(\nhat) = \text{FI-C-9}}_{\Po\text{-odd},\ \Te\text{-even}}.
\]
Consequently:
\begin{enumerate}[label=(\roman*),leftmargin=2.2em]
\item the \emph{entire} $\Po$-odd (spatial-chirality / pseudoscalar) content of $\scyc$ is
$\sgn(\nhat) = \text{FI-C-9}$ --- and by Lemma~\ref{lem:achiral} this is the unique primitive
pseudoscalar, so the identification is forced, not chosen;
\item the \emph{entire} $\Po$-even, $\Te$-odd (temporal-arrow) content of $\scyc$ is $\sgn(\delta)$.
\end{enumerate}
\end{theorem}

\begin{proof}
The symmetry characters are Lemmas~\ref{lem:nhat}--\ref{lem:sigma}. Each side of
\eqref{eq:merge1-relation} carries a well-defined $(\Po,\Te)$ character; the right side is
$(\Po,\Te) = (\text{even}\cdot\text{odd},\ \text{odd}\cdot\text{even}) = (\text{odd},\text{odd})$,
matching the left side (Lemma~\ref{lem:sigma}), so \eqref{eq:merge1-relation} is $\Po,\Te$-covariant.
A product of a $\Po$-even factor and a $\Po$-odd factor has its $\Po$-odd content carried
\emph{entirely} by the $\Po$-odd factor; here that factor is $\sgn(\nhat)$, giving (i). Dually, its
$\Te$-odd content with $\Po$-even character is carried entirely by the $\Po$-even/$\Te$-odd factor
$\sgn(\delta)$, giving (ii). Uniqueness of the pseudoscalar is Lemma~\ref{lem:achiral}.
\end{proof}

\begin{corollary}[The chirality primitive count is ONE --- verdict M1-$\chi$]
\label{cor:count}
The pseudoscalar (spatial-chirality) content of the temporal cycle handedness $\scyc$ equals
$\text{FI-C-9}$, the \emph{same} enantiomorph that THEO-CHIR-CAP-1 pinned the spatial capture
handedness to ($\scap = \zeta\times\text{FI-C-9}$, with $\zeta$ a $\Po$-even registered-geometric
involution). Both faces of chirality --- spatial capture ($\zeta\times\text{FI-C-9}$) and temporal
cycle ($\sgn(\delta)\times\text{FI-C-9}$) --- are the \emph{one} primitive FI-C-9 modulated by a
$\Po$-even factor (a geometric involution $\zeta$ / the temporal arrow $\sgn(\delta)$). The
programme therefore carries \textbf{one} chirality (pseudoscalar) primitive, FI-C-9. This is
\textbf{M1 at the chirality level (M1-$\chi$)}: the chirality-count half of MERGE-$\beta$ is
resolved in the unifying (one-sign) direction.
\end{corollary}

\begin{corollary}[Correction of MERGE-1's M1 route: a parity category-mismatch]
\label{cor:mismatch}
MERGE-1's stated Layer-4 route to M1 --- ``tie $\sgn(\delta)$ to $\sgn(\nhat)$'' --- cannot be
executed as phrased: $\sgn(\delta)$ is $\Po$-even/$\Te$-odd (Lemmas~\ref{lem:delta-p},

\begin{corollary}[The chirality merge is F.2-independent; F.2 / 2a is the pure $\Te$-arrow question]
\label{cor:f2}
MERGE-1 gated MERGE-$\beta$ on both the Layer-4 derivation and the F.2 substrate-Wigner-Eckart

\end{corollary}
