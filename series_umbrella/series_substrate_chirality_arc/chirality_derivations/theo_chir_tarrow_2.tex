\section{The Mechanism-A process and the order-counting}
Mechanism~A posits the DI-bit propagation rate $r(\ehat)=\rzero(1+\delta\,\ehat\cdot\nhat)$ along a unit
edge-direction $\ehat$, with $\nhat$ the substrate primitive (vertex-aligned Reading~C) and
$|\delta|\ll1$. On the 600-cell edge graph these are the transition rates of a continuous-time Markov
process. Its stationary measure is reversible (detailed balance) iff the \textbf{Kolmogorov cycle
condition} holds: for every cycle, the product of forward rates equals the product of backward rates.
The per-directed-edge log-ratio is
\begin{equation}
L(a\to b)=\log\frac{r(a\to b)}{r(b\to a)}=\log\frac{1+\delta c}{1-\delta c}
=2\Bigl(\delta c+\tfrac{\delta^3}{3}c^3+\tfrac{\delta^5}{5}c^5+\cdots\Bigr),\qquad c=\ehat_{ab}\cdot\nhat .
\label{eq:logratio}
\end{equation}

\begin{lemma}[Odd-order structure]\label{lem:odd}
$L$ is odd in $\delta$ (equivalently, $L(b\to a)=-L(a\to b)$, since reversing the edge sends $c\to-c$).
Hence the detailed-balance violation around any cycle contains \emph{only odd powers of $\delta$}; in
particular the $\bigO{\delta^2}$ contribution is identically zero, and the first possible violation is
$\bigO{\delta^3}$.
\end{lemma}
\noindent This corrects the order-counting of the Patch-0688 scoping sketch (which located the gate at
$\bigO{\delta^2}$): there is no $\bigO{\delta^2}$ term.

\section{The per-face triple-product formula}
The triangular faces of the 600-cell generate the cycle space of its $1$-skeleton (the polytope boundary
is a simply-connected triangulated $3$-sphere), so detailed balance holds iff $\sum L$ vanishes around
every triangular face.

\begin{lemma}[Per-face content]\label{lem:perface}
Let a triangular face have oriented edge-projections $a,b,c$ ($c_i=\ehat_i\cdot\nhat$ around the loop).
Then $a+b+c=0$, the $\bigO{\delta^1}$ cycle sum vanishes for every face, and the $\bigO{\delta^3}$ cycle
sum equals $2\delta^3\,abc$.
\end{lemma}
\begin{proof}
The three oriented edge \emph{vectors} sum to zero (the loop closes); all 600-cell edges share the single
length $\ell=1/\varphi$, so $\ell(\ehat_1+\ehat_2+\ehat_3)=0\Rightarrow a+b+c=0$, giving the vanishing
$\bigO{\delta^1}$ term $2\delta(a+b+c)$. By~\eqref{eq:logratio} the $\bigO{\delta^3}$ cycle sum is
$\tfrac{2}{3}\delta^3(a^3+b^3+c^3)$; with $a+b+c=0$ the identity $a^3+b^3+c^3=3abc$ gives
$2\delta^3\,abc$.
\end{proof}

\noindent Thus detailed balance to $\bigO{\delta^3}$ $\iff$ $abc=0$ for every triangular face --- a
finite, concrete condition.

\section{The computation: non-reversibility at $\bigO{\delta^3}$}
\begin{theorem}[$\bigO{\delta^3}$ non-reversibility]\label{thm:main}
Under Mechanism~A at vertex-aligned Reading~C, the induced Markov process on the 600-cell violates the
Kolmogorov cycle condition at $\bigO{\delta^3}$: the per-face content $2\delta^3 abc$ is nonzero on
$420$ of the $1200$ triangular faces (with $|abc|\in\{1/8,1/4\}$), all of which touch the second shell.
The process is therefore non-reversible; it carries a nonzero steady-state probability current.
\end{theorem}
\begin{proof}[Computation]
The verification script \texttt{code/verify\_odelta3\_kolmogorov\_curl.py} constructs the 600-cell ($120$
vertices, $720$ edges, $1200$ triangular $3$-cliques, uniform degree $12$ --- CHECK~1), confirms
$\max_{\text{faces}}|a+b+c|=2.2\times10^{-16}$ and the identity $a^3+b^3+c^3=3abc$ to
$5.6\times10^{-16}$ (CHECK~2), and evaluates $abc$ per face (CHECK~3): $\max|abc|=1/4$, with $420$ faces
nonzero, all incident to the second shell. The first-shell faces contribute zero --- icosahedral
first-shell faces have all edges perpendicular to $\nhat$, and host-side faces have two host-edges of
opposite orientation whose cubes cancel --- so the residual lives exactly at the second shell, as
predicted by the Patch-0688 shell analysis. A single nonzero face suffices to break reversibility, so the
conclusion is stable against higher odd orders.
\end{proof}

\begin{remark}[Why third order]
The bias $\delta\,\ehat\cdot\nhat$ is, per step, the gradient of the linear function $x\cdot\nhat$, hence
conservative --- which is why the $\bigO{\delta^1}$ circulation vanishes. But the Kolmogorov condition
lives in $\log r$, which is \emph{nonlinear} in $\ehat\cdot\nhat$; the $\delta^3$ term $\propto
(\ehat\cdot\nhat)^3$ is not a gradient and carries circulation. The substrate arrow is thus a
\emph{nonlinear} effect of the multiplicative rate law, invisible until the second shell at third order.
\end{remark}
