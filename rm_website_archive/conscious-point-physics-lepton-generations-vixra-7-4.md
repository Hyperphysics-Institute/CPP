<p>def lepton_mass(vertices=4):<br />
    base = 0.511 * (vertices / 4)  # Scaled compression<br />
    phase_factor = np.random.choice(range(1,4)) / cage_layers<br />
    correction = 0.01 * vertices * phase_factor<br />
