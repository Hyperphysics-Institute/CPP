## §7. Embedded verification code (run it → SCRIPT-EXECUTED; Python stdlib only)

This combines the SF-6 Tier-1 inter-constant verifier (`code/1600_verify_sf6_core.py`) and the 1602 band-top transcription check (`code/1602_verify_bandtop.py`). It asserts ONLY the Tier-1 relations + confirms the imported band top — it does **not** assert the Tier-2 toy-model constants as derivations (that would misrepresent the rigor floor the paper states).

```python
# SF-6 review verify — Python stdlib only
import math

# ---- CODATA 2018 reference values ----
C      = 299_792_458.0          # m/s, exact
MU0    = 1.25663706212e-6       # N/A^2
EPS0   = 8.8541878128e-12       # F/m
Z0     = 376.730313668          # ohm
HBAR   = 1.054571817e-34        # J s
H      = 6.62607015e-34         # J s, exact
M_E    = 9.1093837015e-31       # kg
T_P    = 5.391247e-44           # s, Planck time

def rc(a, b, rtol): return abs(a-b) <= rtol*abs(b)

print("=== SF-6 Tier-1 inter-constant identities (must be parameter-free) ===")
c_chk  = 1.0/math.sqrt(MU0*EPS0);  print(f"c = 1/sqrt(mu0 eps0)   = {c_chk:.9g}   (CODATA {C:.9g})");  assert rc(c_chk,C,1e-6)
z_chk  = math.sqrt(MU0/EPS0);      print(f"Z0 = sqrt(mu0/eps0)    = {z_chk:.9g}   (CODATA {Z0:.9g})");  assert rc(z_chk,Z0,1e-6)
mu_chk = Z0/C;                     print(f"mu0  = Z0/c            = {mu_chk:.9g}   (CODATA {MU0:.9g})"); assert rc(mu_chk,MU0,1e-6)
ep_chk = 1.0/(Z0*C);              print(f"eps0 = 1/(Z0 c)        = {ep_chk:.9g}   (CODATA {EPS0:.9g})"); assert rc(ep_chk,EPS0,1e-6)
h_chk  = 2*math.pi*HBAR;          print(f"h = 2 pi hbar          = {h_chk:.9g}   (CODATA {H:.9g})");    assert rc(h_chk,H,1e-6)
mc2    = M_E*C**2; nu_C = mc2/HBAR; mc2_rt = HBAR*nu_C
print(f"mc^2 = hbar nu_C       = {mc2_rt:.9g}   (round-trip {mc2:.9g})");                               assert rc(mc2_rt,mc2,1e-12)

print()
print("=== TP-1 band top imported into SF-6 (transcription check, NOT an SF-6 result) ===")
assert rc(math.sqrt(12.0), 2*math.sqrt(3.0), 1e-12)
omega_max = math.sqrt(12.0)/T_P
omega_g   = 2*math.pi*C/500e-9        # optical ~500 nm
ln_ratio  = math.log(omega_max/omega_g)
print(f"sqrt(12) = 2 sqrt(3)   = {math.sqrt(12.0):.6f}")
print(f"omega_max = sqrt(12)/tP = {omega_max:.3e} rad/s")
print(f"ln(omega_max/omega_g)   = {ln_ratio:.2f}   (TP-1 quotes ~64.5)")
assert 63.0 <= ln_ratio <= 66.0

print()
```
